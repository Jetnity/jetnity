function ack(rows: PgRows): Ack {
  assert.equal(rows.length, 1); assert.ok(typeof rows[0]!.result === 'string' && rows[0]!.result.length < 8192)
  return JSON.parse(rows[0]!.result) as Ack
}
function values(command: Command): readonly (string | number | null)[] {
  const b = command.binding, e = command.expected
  if (command.action === 'stage') return [command.operation, b.methodId, b.revision, b.methodDigest, b.packageDigest, b.revision === 1 ? null : b.revision - 1]
  if (command.action === 'revoke') return [command.operation, b.methodId, b.revision, b.methodDigest, b.packageDigest, e?.epoch ?? null]
  return [command.operation, b.methodId, b.revision, b.methodDigest, b.packageDigest, e?.revision ?? null,
    e?.methodDigest ?? null, e?.packageDigest ?? null, e?.epoch ?? 0, e?.status ?? null, command.mode ?? null, command.until ?? null]
}
function confirmationValues(command: Command): readonly (string | number | null)[] {
  const b = command.binding, e = command.expected
  return [command.operation, command.action, b.methodId, b.revision, b.methodDigest, b.packageDigest,
    command.action === 'stage' && b.revision > 1 ? b.revision - 1 : null,
    e?.revision ?? null, e?.methodDigest ?? null, e?.packageDigest ?? null,
    command.action === 'stage' ? null : e?.epoch ?? 0, e?.status ?? null, command.mode ?? null, command.until ?? null]
}
const queryFor = (command: Command) => command.action === 'stage' ? STAGE_SQL : command.action === 'approve' ? APPROVE_SQL : REVOKE_SQL
const invoke = (connection: LocalPgConnection, command: Command) => connection.query(queryFor(command), parameters(values(command)))
const confirm = async (connection: LocalPgConnection, command: Command) => ack(await connection.query(CONFIRM_SQL, parameters(confirmationValues(command))))

async function installCandidate(connection: LocalPgConnection, t: TestContext): Promise<void> {
  const marker = 'DO $ma1_exact_boundary$'
  const at = candidate.indexOf(marker)
  assert.ok(at > 0); assert.equal(candidate.indexOf(marker, at + marker.length), -1)
  const prefix = candidate.slice(0, at), suffix = candidate.slice(at)
  assert.equal(prefix + suffix, candidate, 'fixed diagnostic split cannot alter candidate bytes')
  try {
    await connection.query(prefix)
    // Same existing connection, still inside the candidate's exact BEGIN/COMMIT.
    const definitions = await connection.query(CHECK_DEFINITIONS)
    assert.equal(definitions.length, 3)
    assert.ok(Buffer.byteLength(JSON.stringify(definitions)) < 32768)
    t.diagnostic('MA1 pre-validator exact CHECK definitions (DDL diagnostics only): ' + JSON.stringify(definitions))
    await connection.query(suffix)
  } catch (error) { await connection.query('ROLLBACK'); throw error }
}

async function until(lifetime: OwnedTestLifetime, predicate: () => Promise<boolean>, message: string, milliseconds = 3000): Promise<void> {
  const deadline = performance.now() + milliseconds
  do { if (await predicate()) return; await lifetime.wait(10) } while (performance.now() < deadline)
  assert.fail(message)
}
function observe<T>(promise: Promise<T>) {
  let settled = false
  const result = promise.then(value => { settled = true; return { value, error: undefined } }, error => { settled = true; return { value: undefined, error } })
  return { result, isSettled: () => settled }
}

async function applyRollback(connection: LocalPgConnection): Promise<void> {
  try { await connection.query(rollback) }
  catch (error) {
    try { await connection.query('ROLLBACK') }
    catch (cleanup) { throw new AggregateError([error, cleanup], 'ma1 rollback refusal cleanup failed') }
    throw error
  }
}

function ipcRecord(value: unknown, keys: readonly string[]): value is Record<string, unknown> {
  if (!value || typeof value !== 'object' || Object.getPrototypeOf(value) !== Object.prototype) return false
  const names = Reflect.ownKeys(value)
  return names.length === keys.length && names.every(key => typeof key === 'string' && keys.includes(key)
    && Object.getOwnPropertyDescriptor(value, key)?.get === undefined)
}
function startWorker(native: Native, lifetime: OwnedTestLifetime, children: Set<ChildProcess>,
  track: <T>(promise: Promise<T>) => Promise<T>) {
  const generation = randomBytes(8).toString('hex'), binding = ma1RootVectors()[0]!.binding
  const child = fork(fileURLToPath(new URL('./process-worker.ts', import.meta.url)), [generation], {
    execArgv: ['--import', fileURLToPath(new URL('../../server-only-test-register.mjs', import.meta.url)), '--import', 'tsx'],
    stdio: ['ignore', 'pipe', 'pipe', 'ipc'], serialization: 'json',
  })
  children.add(child)
  let closing = false, joined = false, fatal: Error | undefined, commandId = 0, lastReadId = 0, outputBytes = 0
  type WorkerResult = Readonly<{ status: string; retainedReceipt: boolean }>
  let command: { id: number; kind: string; resolve: (value: WorkerResult) => void; reject: (error: Error) => void } | undefined
  const reads = new Map<number, AbortController>(), jobs = new Set<Promise<void>>()
  const exit = track(new Promise<{ code: number | null; signal: NodeJS.Signals | null }>(resolve => {
    child.once('exit', (code, signal) => {
      children.delete(child); resolve({ code, signal })
      if (!closing) stop(Error('ma1 unexpected child exit'))
    })
  }))
  function stop(error?: Error): void {
    if (error && !fatal) fatal = error
    closing = true
    for (const controller of reads.values()) controller.abort(error ?? Error('ma1 child lifetime ended'))
    command?.reject(error ?? Error('ma1 child terminated')); command = undefined
    if (child.exitCode === null && child.signalCode === null) child.kill('SIGKILL')
  }
  const timer = setTimeout(() => stop(Error('ma1 child scenario exceeded 60000ms')), 60_000)
  const removeStop = lifetime.onStop(() => stop(Error('ma1 parent lifetime stopped')))
  const send = (frame: Record<string, unknown>) => {
    if (Buffer.byteLength(JSON.stringify(frame)) > 4096 || !child.connected) { stop(Error('ma1 outbound frame unavailable')); return }
    child.send(frame, error => { if (error) stop(error) })
  }
  child.on('error', stop)
  child.on('disconnect', () => { if (!closing) stop(Error('ma1 unexpected child disconnect')) })
  const output = (bytes: Buffer) => { outputBytes += bytes.length; if (outputBytes > 65536) stop(Error('ma1 child output bound exceeded')) }
  child.stdout?.on('data', output); child.stderr?.on('data', output)
  const receive = (input: unknown) => {
    if (closing) return
    try {
      assert.ok(input && typeof input === 'object' && Buffer.byteLength(JSON.stringify(input)) <= 4096)
      const frame = input as Record<string, unknown>
      assert.equal(frame.v, 1); assert.equal(frame.generation, generation)
      if (frame.kind === 'result') {
        assert.ok(ipcRecord(frame, ['v', 'generation', 'commandId', 'kind', 'status', 'retainedReceipt']))
        assert.ok(command && frame.commandId === command.id && typeof frame.retainedReceipt === 'boolean')
        const allowed = command.kind === 'resolve' ? ['approved', 'not_approved', 'unavailable']
          : command.kind === 'reprove_retained' ? ['current', 'revoked', 'expired', 'unavailable']
            : command.kind === 'stop' ? ['stopped'] : ['unavailable']
        assert.ok(typeof frame.status === 'string' && allowed.includes(frame.status))
        const pending = command; command = undefined
        pending.resolve(Object.freeze({ status: frame.status, retainedReceipt: frame.retainedReceipt }))
        return
      }
      if (frame.kind === 'abort') {
        assert.ok(ipcRecord(frame, ['v', 'generation', 'readId', 'kind']) && Number.isSafeInteger(frame.readId))
        const controller = reads.get(frame.readId as number); assert.ok(controller)
        controller.abort(Error('ma1 child requested read abort')); return
      }
      assert.ok(ipcRecord(frame, ['v', 'generation', 'readId', 'kind', 'methodId', 'revision', 'methodDigest', 'packageDigest', 'checkedAt']))
      assert.equal(frame.kind, 'read'); assert.ok(Number.isSafeInteger(frame.readId)
        && (frame.readId as number) > lastReadId && (frame.readId as number) <= 65535 && reads.size < 4)
      const parsed = readMa1CurrentRequest({ methodId: frame.methodId, revision: frame.revision,
        methodDigest: frame.methodDigest, packageDigest: frame.packageDigest, checkedAt: frame.checkedAt })
      assert.ok(parsed.ok)
      if (!parsed.ok) throw Error('ma1 invalid child request')
      assert.deepEqual({ methodId: parsed.value.methodId, revision: parsed.value.revision,
        methodDigest: parsed.value.methodDigest, packageDigest: parsed.value.packageDigest }, binding)
      const id = frame.readId as number; lastReadId = id
      const controller = new AbortController(); reads.set(id, controller)
      const job = track((async () => {
        const result = await native.executor.readCurrent(parsed.value, controller.signal)
        if (result.status === 'rows') assert.ok(readMa1CurrentRows(result.rows, binding, parsed.value.checkedAt).ok)
        if (!closing) send({ v: 1, generation, readId: id, kind: 'settled', result })
      })())
      jobs.add(job)
      void job.then(() => { reads.delete(id); jobs.delete(job) }, error => { reads.delete(id); jobs.delete(job); stop(error instanceof Error ? error : Error('ma1 parent executor failed')) })
    } catch (error) { stop(error instanceof Error ? error : Error('ma1 invalid IPC frame')) }
  }
  child.on('message', receive)
  async function issue(kind: 'resolve' | 'reprove_retained' | 'try_serialized_clone' | 'stop', checkedAt = now()): Promise<WorkerResult> {
    if (closing || command || commandId >= 65535) throw Error('ma1 command boundary unavailable')
    const id = ++commandId
    return await track(new Promise<WorkerResult>((resolve, reject) => {
      command = { id, kind, resolve, reject }
      send({ v: 1, generation, commandId: id, kind, ...(['resolve', 'reprove_retained'].includes(kind) ? { checkedAt } : {}) })
    }))
  }
  async function join(): Promise<{ code: number | null; signal: NodeJS.Signals | null }> {
    const terminal = await exit
    try {
      await lifetime.drain([...jobs])
      assert.equal(reads.size, 0); assert.equal(jobs.size, 0); joined = true
    } finally {
      clearTimeout(timer); removeStop(); child.removeListener('message', receive)
      child.stdout?.removeListener('data', output); child.stderr?.removeListener('data', output)
    }
    if (fatal) throw fatal
    return terminal
  }
  return Object.freeze({ generation, child, issue,
    pendingReads() { return reads.size },
    async killAndJoin() { stop(); const terminal = await join(); assert.equal(terminal.signal, 'SIGKILL'); return terminal },
    async dispose() { if (!joined) { stop(); await join() } },
    async close() {
      if (joined) return
      if (!closing) {
        // Expected stop reply is received while the channel remains live; mark
        // closing synchronously when that exact reply arrives before disconnect.
        const stopPromise = issue('stop')
        const stopFrame = (input: unknown) => {
          if (ipcRecord(input, ['v', 'generation', 'commandId', 'kind', 'status', 'retainedReceipt'])
            && input.generation === generation && input.status === 'stopped') closing = true
        }
        child.on('message', stopFrame)
        try { assert.equal((await stopPromise).status, 'stopped') } finally { child.removeListener('message', stopFrame) }
      }
      const terminal = await join(); assert.equal(terminal.code, 0)
    },
  })
}

    await t.test('rollback takes all removal locks before checking concurrently committed history', async () => {
      const concurrent = await make(), binding = ma1SyntheticTuple('rollback-wait')

      const at = rollback.indexOf(lockSql)
      assert.ok(at > 0); assert.equal(rollback.indexOf(lockSql, at + lockSql.length), -1)
      const prefix = rollback.slice(0, at), suffix = rollback.slice(at + lockSql.length)
      assert.equal(prefix + lockSql + suffix, rollback, 'fixed lock observation split preserves the complete original rollback bytes')
      let pendingLock: ReturnType<typeof observe<PgRows>> | undefined
      try {
        await mutator.query('BEGIN'); const command = await concurrent.stage(mutator, binding)
        const beforeCatalog = await catalogSnapshot(concurrent)
        await remover.query(prefix)
        pendingLock = observe(tracked(remover.query(lockSql)))
        await waitBlocked(concurrent, lifetime, concurrent.identities.get(remover)!, concurrent.identities.get(mutator)!, lockSql, () => !pendingLock!.isSettled())
        await mutator.query('COMMIT')
        const locked = await pendingLock.result; assert.ifError(locked.error)
        await assert.rejects(remover.query(suffix), code('P7507'))
        await remover.query('ROLLBACK')
        const state = await concurrent.state(binding)
        assert.equal(JSON.parse(state[0]!.revisions!).length, 1); assert.equal(JSON.parse(state[0]!.operations!).length, 1)
        assert.equal((await confirm(mutator, command)).status, 'recorded')
        assert.deepEqual(await catalogSnapshot(concurrent), beforeCatalog, 'post-wait refusal removes no candidate metadata')
      } finally {
        // Same connections and original transaction: any prefix/lock/suffix error
        // joins rollback and the real pending lock; no alternate removal path.
        await mutator.query('ROLLBACK')
        if (pendingLock) await pendingLock.result
        await remover.query('ROLLBACK'); mutator.close(); remover.close()
      }
    })
