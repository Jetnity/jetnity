type Identity = Readonly<{ pid: string; birth: string }>

type Command = Readonly<{ operation: string; action: 'stage' | 'approve' | 'revoke'; binding: Ma1MethodBinding;
  expected?: Expected; mode?: 'until' | 'until_revoked'; until?: string }>
type Ack = Readonly<{ status: string; operationId: string; action?: string; actorId?: string;

let operationIndex = 0
const op = () => ma1Operation(++operationIndex)

const parameters = (values: readonly (string | number | null)[]) => values.map(value => pgText(value === null ? '' : String(value)))

const code = (state: string) => (error: unknown) => error instanceof LocalPgError && error.code === state
function ack(rows: PgRows): Ack {
  assert.equal(rows.length, 1); assert.ok(typeof rows[0]!.result === 'string' && rows[0]!.result.length < 8192)
  return JSON.parse(rows[0]!.result) as Ack
}
function values(command: Command): readonly (string | number | null)[] {
  const b = command.binding, e = command.expected
  if (command.action === 'stage') return [command.operation, b.methodId, b.revision, b.methodDigest, b.packageDigest, b.revision === 1 ? null : b.revision - 1]
  if (command.action === 'revoke') return [command.operation, b.methodId, b.revision, b.methodDigest, b.packageDigest, e?.epoch ?? null]
  return [command.operation, b.methodId, b.revision, b.methodDigest, b.packageDigest, e?.revision ?? null,
    e?.methodDigest ?? null, e?.packageDigest ?? null, e?.epoch ?? 0, e?.status ?? null, command.mode ?? null, command.until ?? null]
}
function confirmationValues(command: Command): readonly (string | number | null)[] {
  const b = command.binding, e = command.expected
  return [command.operation, command.action, b.methodId, b.revision, b.methodDigest, b.packageDigest,
    command.action === 'stage' && b.revision > 1 ? b.revision - 1 : null,
    e?.revision ?? null, e?.methodDigest ?? null, e?.packageDigest ?? null,
    command.action === 'stage' ? null : e?.epoch ?? 0, e?.status ?? null, command.mode ?? null, command.until ?? null]
}
const queryFor = (command: Command) => command.action === 'stage' ? STAGE_SQL : command.action === 'approve' ? APPROVE_SQL : REVOKE_SQL
const invoke = (connection: LocalPgConnection, command: Command) => connection.query(queryFor(command), parameters(values(command)))
const confirm = async (connection: LocalPgConnection, command: Command) => ack(await connection.query(CONFIRM_SQL, parameters(confirmationValues(command))))

async function installCandidate(connection: LocalPgConnection, t: TestContext): Promise<void> {
  const marker = 'DO $ma1_exact_boundary$'
  const at = candidate.indexOf(marker)
  assert.ok(at > 0); assert.equal(candidate.indexOf(marker, at + marker.length), -1)
  const prefix = candidate.slice(0, at), suffix = candidate.slice(at)
  assert.equal(prefix + suffix, candidate, 'fixed diagnostic split cannot alter candidate bytes')
  try {
    await connection.query(prefix)
    // Same existing connection, still inside the candidate's exact BEGIN/COMMIT.
    const definitions = await connection.query(CHECK_DEFINITIONS)
    assert.equal(definitions.length, 3)
    assert.ok(Buffer.byteLength(JSON.stringify(definitions)) < 32768)
    t.diagnostic('MA1 pre-validator exact CHECK definitions (DDL diagnostics only): ' + JSON.stringify(definitions))
    await connection.query(suffix)
  } catch (error) { await connection.query('ROLLBACK'); throw error }
}

type Fixture = Awaited<ReturnType<typeof createFixture>>

async function until(lifetime: OwnedTestLifetime, predicate: () => Promise<boolean>, message: string, milliseconds = 3000): Promise<void> {
  const deadline = performance.now() + milliseconds
  do { if (await predicate()) return; await lifetime.wait(10) } while (performance.now() < deadline)
  assert.fail(message)
}
function observe<T>(promise: Promise<T>) {
  let settled = false
  const result = promise.then(value => { settled = true; return { value, error: undefined } }, error => { settled = true; return { value: undefined, error } })
  return { result, isSettled: () => settled }
}
async function waitBlocked(f: Fixture, lifetime: OwnedTestLifetime, caller: Identity, holder: Identity,
  expectedSql: string, pending: () => boolean, relation = 'official_v3_method_private.heads_v1'): Promise<void> {
  await until(lifetime, async () => {
    const rows = await f.observer.query(`SELECT a.state,a.wait_event_type,a.query,
      ($3::integer=ANY(pg_catalog.pg_blocking_pids(a.pid)))::text AS actual_holder,
      EXISTS(SELECT 1 FROM pg_catalog.pg_locks l WHERE l.pid=a.pid AND l.relation=$4::regclass)::text AS relevant_relation
      FROM pg_catalog.pg_stat_activity a WHERE a.pid=$1::integer AND a.backend_start=$2::timestamptz`,
    [pgText(caller.pid, 23), pgText(caller.birth), pgText(holder.pid, 23), pgText(relation)])
    return rows.length === 1 && rows[0]!.state === 'active' && rows[0]!.wait_event_type === 'Lock'
      && rows[0]!.query === expectedSql && rows[0]!.actual_holder === 'true' && rows[0]!.relevant_relation === 'true' && pending()
  }, 'actual PID/birth/query/Lock/blocker/relation/pending proof missing')
}
async function assertBothAbsent(f: Fixture, native: Native): Promise<void> {
  native.assertHealthy()
  for (const entry of native.evidence.calls) {
    assert.ok(entry.joined && entry.pending === 0 && entry.targetAbsent && entry.controlAbsent && !entry.stoppedFailure)
    for (const id of [entry.target!, entry.control!]) {
      const rows = await f.observer.query('SELECT count(*)::text AS n FROM pg_catalog.pg_stat_activity WHERE pid=$1::integer AND backend_start=$2::timestamptz',
        [pgText(id.pid, 23), pgText(id.birth)])
      assert.equal(rows[0]!.n, '0')
    }
  }
}

      const secondResult = await second.result; assert.ifError(secondResult.error); assert.deepEqual(ack(secondResult.value!), first)
      const expected = await f.head(binding), nextA: Command = { action: 'approve', operation: op(), binding, expected, mode: 'until_revoked' }
      const nextB: Command = { ...nextA, operation: op() }
      await a.query('BEGIN'); await invoke(a, nextA)
      const competing = observe(tracked(invoke(b, nextB)))
      await waitBlocked(f, lifetime, f.identities.get(b)!, f.identities.get(a)!, APPROVE_SQL, () => !competing.isSettled())
      await a.query('COMMIT'); assert.ok(code('P7503')((await competing.result).error))
      assert.equal((await f.head(binding)).epoch, expected.epoch + 1)
      a.close(); b.close()
    })

    await t.test('lost response confirms immutable truth across expiry, revoke and supersession', async () => {
      const binding = ma1SyntheticTuple('lost-ack'), revision2 = ma1SyntheticTuple('lost-ack', 2)
      await f.stage(human, binding)
      const expiry = new Date(Date.now() + 800).toISOString()
      const command: Command = { operation: op(), action: 'approve', binding, mode: 'until', until: expiry }
      const caller = await f.human(); await caller.query('BEGIN'); await invoke(caller, command); await caller.query('COMMIT')
      assert.ok(caller.operationDiagnostics.some(entry => entry.phase === 'commit' && entry.commitAcknowledged))
      // Deliberately discard the returned record only AFTER actual committed backend completion.
      caller.close()
      await f.owner.query('SELECT pg_catalog.pg_sleep(GREATEST(0::double precision,extract(epoch FROM ($1::timestamptz-clock_timestamp()))::double precision)+0.02)', [pgText(expiry)])
      const confirmed = await confirm(human, command); assert.equal(confirmed.status, 'recorded')
      await f.revoke(human, binding)
      assert.deepEqual(await confirm(human, command), confirmed)
      await f.stage(human, revision2); await f.approve(human, revision2, await f.head(binding))
      const current = await f.state(binding)
      assert.deepEqual(ack(await invoke(human, command)), confirmed)
      assert.deepEqual(await confirm(human, command), confirmed)
      assert.deepEqual(await f.state(binding), current, 'historical ACK cannot restore authority')
      const unknown = { ...command, operation: op() }
      assert.deepEqual(await confirm(human, unknown), { status: 'absent', operationId: unknown.operation })
      assert.deepEqual(await f.state(binding), current, 'serialized absence writes no history despite stale NEW-state expiry/CAS')
      const backward: Command = { action: 'approve', operation: op(), binding, expected: await f.head(binding), mode: 'until_revoked' }
      await assert.rejects(invoke(human, backward), code('P7503'))
      assert.deepEqual(await f.state(binding), current)
    })

    await t.test('last approval retains a terminal revoke; epoch exhaustion never resets history', async () => {
      const binding = ma1SyntheticTuple('terminal')
      await f.stage(human, binding); await f.approve(human, binding)
      await f.owner.query('UPDATE official_v3_method_private.heads_v1 SET epoch=2147483645 WHERE method_id=$1', [pgText(binding.methodId)])
      t.diagnostic('Epoch2147483645 is explicit synthetic owner setup; no billions-of-operations history claim.')
      await f.approve(human, binding, await f.head(binding))
      assert.equal((await f.head(binding)).epoch, 2147483646)
      let snapshot = await f.state(binding)
      await assert.rejects(invoke(human, { action: 'approve', operation: op(), binding, expected: await f.head(binding), mode: 'until_revoked' }), code('P7503'))
      assert.deepEqual(await f.state(binding), snapshot)
      const terminal = await f.revoke(human, binding)
      assert.equal((await f.head(binding)).epoch, 2147483647)
      const historical = await confirm(human, terminal); snapshot = await f.state(binding)
      await assert.rejects(invoke(human, { action: 'approve', operation: op(), binding, expected: await f.head(binding), mode: 'until_revoked' }), code('P7503'))
      await assert.rejects(invoke(human, { ...terminal, operation: op() }), code('P7503'))
      assert.deepEqual(await confirm(human, terminal), historical)
      assert.deepEqual(await f.state(binding), snapshot)
    })

    // Remaining real lock/capacity/process/rollback groups are appended below.
    assert.deepEqual(await f.owner.query("SELECT proowner::text,proacl::text,prosrc,proconfig::text FROM pg_catalog.pg_proc WHERE oid='jetnity_ot_auth.require_current_owner_session_v1()'::regprocedure"), helperBefore)
  } catch (error) { primaryError = error }
  finally {
    for (const child of children) child.kill('SIGKILL')
    for (const native of natives) native.closeObserver()
    for (const fixture of fixtures) fixture.close()
    try {
      await lifetime.drain([...actors])
      const final = await lifetime.dispose()
      assert.equal(final.pending, 0); assert.equal(final.started, final.settled); assert.equal(final.connections, final.closed)
      t.diagnostic('MA1 owned lifetime terminal evidence: ' + JSON.stringify(final))
    } catch (error) { primaryError = primaryError ? new AggregateError([primaryError, error], 'ma1 primary and cleanup failures') : error }
  }
  if (primaryError) throw primaryError
})
