LITERAL_PINS = {
    'heads_v1_shape': (879, 'd978b0c81b68a7f3bf78e59b3d335543b27d7a8e851857e34c15e92e0ef9e258'),
    'operations_v1_shape': (2800, '64678ff1541d107cf12ced52e99cae1cbd881417087330ebc67930a9fc157a51'),
    'revisions_v1_shape': (280, '60d220d66084557ef4e1a279c2d94a9e5285b2d7426cc6cad05b87f620a20243'),
}
LITERAL_RE = re.compile(rb"\('([^']+)',\$ma1_expected\$(.*?)\$ma1_expected\$\)", re.S)

def sha(data):
    return hashlib.sha256(data).hexdigest()

def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT,
        env={**os.environ, 'GIT_NO_LAZY_FETCH': '1'})

def literals(data):
    matches = list(LITERAL_RE.finditer(data))
    assert len(matches) == 3
    result = {m.group(1).decode(): m for m in matches}
    assert set(result) == set(LITERAL_PINS)
    return result

assert git('rev-parse', 'HEAD').decode().strip() == HEAD
assert git('rev-parse', 'HEAD^{tree}').decode().strip() == TREE
assert git('diff', '--name-only') == b''
assert git('diff', '--cached', '--name-only') == b''
assert sha((DOCS / 'MA1_REVIEW_R2_TASK_V1.md').read_bytes()) == TASK_SHA
r1_bytes = (DOCS / 'MA1_REVIEW_R1_SOURCE_MANIFEST_V1.json').read_bytes()
assert sha(r1_bytes) == R1_SHA
inventory = json.loads(r1_bytes)['currentImplementationInventory']
assert len(inventory) == 13
for entry in inventory:
    assert sha((ROOT / entry['path']).read_bytes()) == entry['sha256']

preserved_docs = {}
for raw in git('ls-files', '-z', 'docs/official-truth-product-v3').split(b'\0'):
    if raw:
        path = raw.decode()
        preserved_docs[path] = sha((ROOT / path).read_bytes())
for path in [DOCS / 'MA1_REVIEW_R2_TASK_V1.md', DOCS / 'MA1_REVIEW_R2_ROOT_FAILURE_EVIDENCE_V1.json',
        *sorted((DOCS / 'ma1-r2-review-evidence').glob('*'))]:
    if path.is_file():
        preserved_docs[path.relative_to(ROOT).as_posix()] = sha(path.read_bytes())

raw_log = LOG.read_bytes()

diagnostics = [(i, line) for i, line in enumerate(raw_log.splitlines(), 1) if MARKER in line]

assert line_number == 33806
actual_rows = json.loads(diagnostic_line.split(MARKER, 1)[1].strip())
assert len(actual_rows) == 3
actual = {row['conname']: row['expression'].encode() for row in actual_rows}
assert set(actual) == set(LITERAL_PINS)
for name, value in actual.items():
    assert (len(value), sha(value)) == LITERAL_PINS[name]

planned = []
for path, (before_sha, after_sha) in SOURCES.items():
    before = (ROOT / path).read_bytes()
    assert sha(before) == before_sha
    assert git('show', HEAD + ':' + path) == before
    old_literals = literals(before)
    comparisons = []
    for name, match in old_literals.items():
        value = match.group(2)
        if name == 'operations_v1_shape':
            assert len(value) == 2801
            assert sha(value) == '66cdd04deb21243789325f646de1a7687b47346aee3de2a25627efde6b9064d3'
            assert value[1643:1644] == b')'
            assert value[:1643] + value[1644:] == actual[name]
            assert value.count(b"until'::text)))) AND ((before_state") == 1
        else:
            assert value == actual[name]
        comparisons.append({'name': name, 'beforeBytes': len(value), 'beforeSha256': sha(value),
            'actualBytes': len(actual[name]), 'actualSha256': sha(actual[name]),
            'equalBefore': value == actual[name]})
    offset = old_literals['operations_v1_shape'].start(2) + 1643
    # Fixed, reviewed one-byte source edit; never construct SQL from CI definitions.
    after = before[:offset] + before[offset + 1:]
    assert sha(after) == after_sha
    assert len(before) - len(after) == 1
    assert before[offset:offset + 1] == b')'
    assert all(m.group(2) == actual[name] for name, m in literals(after).items())
    planned.append({'path': path, 'before': before, 'after': after, 'offset': offset,
        'comparisons': comparisons})

# All inputs and both complete proposed files were checked before either write.

    (ROOT / item['path']).write_bytes(item['after'])

for item in planned:
    final = (ROOT / item['path']).read_bytes()

    assert all(m.group(2) == actual[name] for name, m in literals(final).items())
