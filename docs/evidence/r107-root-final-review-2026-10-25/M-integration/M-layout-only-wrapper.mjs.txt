async function oeffnenBei(browser, query) {
  const page = await browser.newPage({ viewport: { width: 1280, height: 900 } })
  await page.clock.install({ time: new Date('2026-10-25T12:00:00Z') })
  await page.emulateMedia({ reducedMotion: 'reduce' })
  await page.goto(query ? `${origin}/?${query}` : origin)
  await page.waitForSelector('[data-sammlung-wurzel]')
  return page
}

async function ruhen(page) {
  await page.evaluate(
    () => new Promise((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => resolve(undefined)))),
  )
}

async function ergebnis(page, teil) {

    const fenster = window
    fenster.__sammlung.setErgebnis({ ...fenster.__sammlung.anfang, ...stueck })
  }, teil)
  await ruhen(page)
}
