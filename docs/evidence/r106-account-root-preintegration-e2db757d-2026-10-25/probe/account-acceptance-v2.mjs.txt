function parseOptions() {
  const args = process.argv.slice(2)
  assert.equal(args.length % 2, 0, 'Every option requires a value')
  const options = { only: 'all' }
  const seen = new Set()
  for (let index = 0; index < args.length; index += 2) {
    const name = args[index]
    assert.ok(['--repo', '--expected-head', '--output-dir', '--only'].includes(name), `Unknown option ${name}`)
    assert.ok(!seen.has(name), `Duplicate option ${name}`)
    seen.add(name)
    options[name.slice(2)] = args[index + 1]
  }
  for (const key of ['repo', 'expected-head', 'output-dir']) assert.ok(options[key], `Required --${key}`)
  assert.match(options['expected-head'], /^[a-f0-9]{40}$/, 'An exact lowercase 40-hex expected head is required')
  assert.notEqual(options['expected-head'], DEFECT_HEAD, 'REFUSED: this is the independently demonstrated defective head, not an acceptance candidate')
  assert.ok(['all', 'pure', 'mounted'].includes(options.only), 'Invalid --only value')
  assert.equal(process.versions.node.split('.')[0], '22', 'Use the repository Node 22 runtime')
  assert.ok(isAbsolute(options.repo) && isAbsolute(options['output-dir']), 'Use absolute paths')
  options.repo = realpathSync(options.repo)
  options['output-dir'] = resolve(options['output-dir'])
  assert.ok(!existsSync(options['output-dir']), 'Refusing to overwrite earlier evidence')
  const rel = relative(options.repo, options['output-dir'])
  assert.ok(rel.startsWith(`..${sep}`) || isAbsolute(rel), 'Evidence must remain outside the checkout')
  return options
}

function git(...args) {
  return execFileSync('git', ['-C', repo, ...args], { encoding: 'utf8', stdio: ['ignore', 'pipe', 'pipe'] }).trim()
}
function hashFile(path, algorithm = 'sha256') {
  return createHash(algorithm).update(readFileSync(path)).digest('hex')
}
function gitBlob(path) {
  const bytes = readFileSync(path)
  return createHash('sha1').update(`blob ${bytes.length}\0`).update(bytes).digest('hex')
}
function provenance() {
  assert.equal(realpathSync(git('rev-parse', '--show-toplevel')), repo, 'The supplied path must be the checkout root')
  const head = git('rev-parse', 'HEAD')
  assert.equal(head, expectedHead, 'Checkout HEAD differs from the explicitly expected head')
  assert.equal(git('status', '--porcelain', '--untracked-files=no'), '', 'Tracked files/index must be clean')
  const files = {}
  for (const path of [...sourcePaths].sort()) {
    const committed = git('rev-parse', `${expectedHead}:${path}`)
    assert.match(committed, /^[a-f0-9]{40}$/, `Missing committed blob: ${path}`)
    const actual = gitBlob(join(repo, path))
    assert.equal(actual, committed, `Source bytes are not the expected committed blob: ${path}`)
    files[path] = { gitBlob: actual, sha256: hashFile(join(repo, path)) }
  }
  return { head, trackedCheckoutClean: true, files, checkedAt: new Date().toISOString() }
}

async function check(name, operation) {
  const startedAt = new Date().toISOString()
  try {
    const evidence = await operation()
    report.cases.push({ name, result: 'PASS', startedAt, evidence })
  } catch (error) {
    report.cases.push({ name, result: 'FAIL', startedAt, error: String(error?.stack ?? error) })
  }
}
function recordBundleSources(build) {
  for (const path of Object.keys(build.metafile.inputs)) {
    if (path.startsWith('<')) continue
    const abs = resolve(repo, path)
    const rel = relative(repo, abs)
    if (!rel.startsWith(`..${sep}`) && !isAbsolute(rel) && !rel.split(sep).includes('node_modules')) sourcePaths.add(rel)
  }
  // Verify actual source bytes used for this bundle before executing it.
  return provenance()
}

        // Accept either keeping a successfully saved edit open or closing it; the reopen contract is what matters.
        if (await form.count()) {
          if (kind === 'traveller') await page.locator(`[data-fokus="aendern:${target}"]`).click()
          else await form.getByRole('button', { name: copy.abbrechen, exact: true }).click()
          await expect(form).toHaveCount(0)
        }
        await setProps(page, props(person('Alice', [document()])))
        form = await openEditor()
        if (kind === 'traveller') {
          await expect(labelInput(form)).toHaveValue('Alice')
          await expect(form.getByRole('combobox', { name: copy.wohnsitzLabel, exact: true })).toHaveValue('DE')
        } else {
          await expect(form.getByRole('combobox', { name: copy.dokumentTypLabel, exact: true })).toHaveValue('passport')
          await expect(form.getByRole('combobox', { name: copy.dokumentIssuerLabel, exact: true })).toHaveValue('DE')
          await expect(form.locator('input[type="date"]')).toHaveValue('2030-01-01')
        }
        await submit(page, form, 2)
        const second = (await calls(page))[1]
        const expected = kind === 'traveller'
          ? { id: target, label: 'Alice', residenceCountryCode: 'DE' }
          : { travellerId: target, documentId: docId, ...documentFields() }
        assert.equal(second.name, kind === 'traveller' ? 'traveller-aendern' : 'document-aendern')
        assert.deepEqual(second.argument, expected)
        const parsed = kind === 'traveller' ? input.registryTravellerAenderungLesen(second.argument) : input.registryDocumentAenderungLesen(second.argument)
        assert.equal(parsed.ok, true)
        return { secondAction: second, frozenParserResult: parsed }
      })
    }

    for (const outcome of ['success', 'reject']) {
      await mounted(`mounted A-R2 explicit adoption survives real late ${outcome}`, props(person()), async (page, copy) => {
        const form = await edit(page)
        await labelInput(form).fill('My newer draft')
        await setProps(page, props(person('Canonical two')))
        await expect(form.getByText(copy.konflikt, { exact: true })).toBeVisible()
        await submit(page, form, 1)
        const epochBefore = Number(await form.getAttribute('data-registry-epoche'))
        await form.getByRole('button', { name: copy.uebernehmen, exact: true }).click()
        await expect(labelInput(form)).toHaveValue('Canonical two')
        const epochAfter = Number(await form.getAttribute('data-registry-epoche'))
        assert.ok(Number.isFinite(epochBefore) && Number.isFinite(epochAfter) && epochAfter > epochBefore)
        await labelInput(form).focus()
        await settle(page, 1, outcome)
        await expect(form).toBeVisible()
        await expect(labelInput(form)).toHaveValue('Canonical two')
        await expect(labelInput(form)).toBeFocused()
        assert.equal(Number(await form.getAttribute('data-registry-epoche')), epochAfter)
        await expect(form.locator('[role="alert"]')).toHaveCount(0)
        const receipt = page.locator(`[data-registry-beleg][data-registry-ausgang="${outcome === 'success' ? 'gespeichert' : 'unbestaetigt'}"]`)
        await expect(receipt).toBeVisible()
        assert.equal(await receipt.evaluate(element => Boolean(element.closest('form'))), false)
        if (outcome === 'reject') await expect(receipt.getByText(copy.unbekannt, { exact: true })).toBeVisible()
        assert.equal((await calls(page)).length, 1)
        return { epochBefore, epochAfter, retainedFields: 'Canonical two', focus: await focus(page), settlements: await page.evaluate(() => window.__accountReview.settlements) }
      })
    }

    await mounted('mounted actual Promise rejection after A to B keeps B draft and focus', props(person(), person('Bob', [], otherTarget)), async (page, copy) => {
      await submit(page, await edit(page), 1)
      await page.locator(`[data-fokus="verwalten:${otherTarget}"]`).click()
      const formB = await edit(page, otherTarget)
      await labelInput(formB).fill('Bob newer unsent draft')
      await labelInput(formB).focus()
      await settle(page, 1, 'reject')
      await expect(labelInput(formB)).toHaveValue('Bob newer unsent draft')
      await expect(labelInput(formB)).toBeFocused()
      await expect(formB.locator('[role="alert"]')).toHaveCount(0)
      await expect(page.getByText(copy.unbekannt, { exact: true })).toBeVisible()
      assert.equal((await calls(page)).length, 1)
      return { targetA: target, targetB: otherTarget, retainedDraft: 'Bob newer unsent draft', actionCount: 1, settlements: await page.evaluate(() => window.__accountReview.settlements) }
    })

    await mounted('mounted A-R4 a newer historical-receipt focus move survives pending cancel restoration', props(person()), async (page, copy) => {
      await submit(page, await edit(page), 1)
      await settle(page, 1, 'reject')
      const receipt = page.locator('[data-registry-beleg][data-registry-ausgang="unbestaetigt"]')
      await expect(receipt.getByText(copy.unbekannt, { exact: true })).toBeVisible()
      const repeat = receipt.getByRole('button', { name: copy.erneutAktion, exact: true })
      await expect(repeat).toBeVisible()
      await page.getByRole('button', { name: copy.staatsbuergerschaftHinzufuegen, exact: true }).click()
      const form = formFor(page, `${surface}:citizenship-anlegen`)
      await expect(form).toBeVisible()
      const immediate = await form.getByRole('button', { name: copy.abbrechen, exact: true }).evaluate(button => {
        button.click()
        const receipt = document.querySelector('[data-registry-beleg][data-registry-ausgang="unbestaetigt"]')
        const newerControl = receipt?.querySelector('button')
        if (!(newerControl instanceof HTMLButtonElement)) throw new Error('Missing already-mounted receipt repeat control')
        newerControl.focus()
        return {
          focusedReceiptControl: document.activeElement === newerControl,
          editor: newerControl.closest('[data-registry-editor]')?.getAttribute('data-registry-editor'),
          name: newerControl.textContent?.trim(),
        }
      })
      assert.equal(immediate.focusedReceiptControl, true, 'The later focus move must actually occur before checking ownership')
      await expect(form).toHaveCount(0)
      await flush(page)
      const afterCommit = await focus(page)
      assert.equal(
        await repeat.evaluate(element => element === document.activeElement),
        true,
        `Newer receipt focus must win over canceled-editor restoration; observed ${JSON.stringify(afterCommit)}`,
      )
      assert.equal((await calls(page)).length, 1, 'Focusing a historical receipt must not repeat its action')
      return { immediate, afterCommit, actionCount: 1, attemptedRepeat: false }
    })

    for (const kind of ['citizenship-create', 'document-create', 'document-edit-at-limit']) {
      const documents = kind === 'document-edit-at-limit'
        ? Array.from({ length: 12 }, (_, index) => document(`00000000-0000-4000-8000-${String(index + 10).padStart(12, '0')}`))
        : []
      await mounted(`mounted A-R4 keyboard cancel ${kind} restores a live control without stealing later focus`, props(person('Alice', documents)), async (page, copy) => {
        const editor = kind === 'citizenship-create' ? `${surface}:citizenship-anlegen` : kind === 'document-create' ? `${surface}:document-anlegen` : documentEditor
        const opener = kind === 'citizenship-create'
          ? page.getByRole('button', { name: copy.staatsbuergerschaftHinzufuegen, exact: true })
          : kind === 'document-create'
            ? page.getByRole('button', { name: copy.dokumentHinzufuegen, exact: true })
            : page.getByRole('button', { name: copy.dokumentAendern, exact: true }).first()
        const openEditor = async () => { await opener.click(); const form = formFor(page, editor); await expect(form).toBeVisible(); return form }
        let form = await openEditor()
        const cancel = form.getByRole('button', { name: copy.abbrechen, exact: true })
        await cancel.focus()
        await page.keyboard.press('Enter')
        await expect(form).toHaveCount(0)
        await flush(page)
        const active = await focus(page)
        assert.equal(active.connected, true)
        assert.equal(active.tag, 'BUTTON', 'Cancel must focus a live button, never BODY or a removed control')
        if (kind !== 'document-edit-at-limit') await expect(opener).toBeFocused()
        else {
          const onOpener = await opener.evaluate(element => element === document.activeElement)
          const manage = page.locator(`[data-fokus="verwalten:${target}"]`)
          const onStableFallback = await manage.evaluate(element => element === document.activeElement)
          assert.ok(onOpener || onStableFallback, 'At the document limit, focus must return to that document edit opener or stable manage control')
          assert.equal(await page.getByRole('button', { name: copy.dokumentHinzufuegen, exact: true }).count(), 0)
        }
        // A newer user focus move in the same turn must win over delayed cancel-focus work.
        form = await openEditor()
        await form.getByRole('button', { name: copy.abbrechen, exact: true }).evaluate(button => {
          button.click()
          document.getElementById('independent-outside').focus()
        })
        await expect(form).toHaveCount(0)
        await flush(page)
        await expect(page.locator('#independent-outside')).toBeFocused()
        assert.equal((await calls(page)).length, 0, 'Cancel must not dispatch an action')
        return { keyboardCancelFocus: active, laterFocus: await focus(page), actionCount: 0 }
      })
    }
  }
  assert.deepEqual(report.pageErrors, [], 'Mounted page errors are not acceptance')
  assert.deepEqual(report.externalRequests, [], 'Any attempted external request invalidates isolated evidence')
} catch (error) {
  report.infrastructureOrContractFailure = String(error?.stack ?? error)
} finally {
  if (browser) await browser.close()
  if (server?.listening) await new Promise(resolve => server.close(resolve))
  try { report.finalProvenance = provenance() }
  catch (error) { report.provenanceFailure = String(error?.stack ?? error) }
  report.endedAt = new Date().toISOString()
  const expectedCases = options.only === 'pure' ? 4 : options.only === 'mounted' ? 10 : 14
  report.expectedCaseCount = expectedCases
  report.actualCaseCount = report.cases.length
  const passed = !report.infrastructureOrContractFailure && !report.provenanceFailure
    && report.cases.length === expectedCases && report.cases.every(entry => entry.result === 'PASS')
    && report.pageErrors.length === 0 && report.externalRequests.length === 0
  report.result = passed ? `BOUNDED_${options.only.toUpperCase()}_ACCEPTANCE_PASS` : 'ACCEPTANCE_NOT_PASSED'
  const output = join(out, 'account-acceptance-result.json')
  writeFileSync(output, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' })
  console.log(JSON.stringify({ result: report.result, expectedHead, output, cases: report.cases.map(({ name, result }) => ({ name, result })), failure: report.infrastructureOrContractFailure ?? report.provenanceFailure }, null, 2))
  if (!passed) process.exitCode = 1
}
