function sourceSnapshot() {
  assert.equal(git('rev-parse', 'HEAD'), expectedHead, 'HEAD changed during probe.')
  assert.equal(git('status', '--porcelain'), '', 'Checkout became dirty.')
  return Object.fromEntries(sources.map(path => {
    const bytes = readFileSync(join(repo, path))
    const gitBlob = createHash('sha1').update(`blob ${bytes.length}\0`).update(bytes).digest('hex')
    assert.equal(gitBlob, git('rev-parse', `${expectedHead}:${path}`), `Working bytes differ from committed head: ${path}`)
    return [path, { gitBlob, sha256: createHash('sha256').update(bytes).digest('hex') }]
  }))
}

function report(board, marker, checkedAt = base) {
  const health = board === 'health'
  const ids = health ? ['app', 'vercel', 'supabase', 'github', 'infomaniak'] : ['provider-ops', 'kill-switch', 'cost-guard', 'model-usage']
  const fresh = ttlMs => ({ state: 'fresh', ageMs: 0, ttlMs })
  const items = ids.map(id => ({
    id, name: id, status: health ? (id === 'app' ? 'unknown' : 'not_configured') : (id === 'model-usage' ? 'empty' : 'foundation_only'),
    source: 'local-independent-get401-acceptance', checkedAt: iso(checkedAt),
    freshness: fresh(id === 'github' || id === 'infomaniak' ? 300_000 : id === 'vercel' || id === 'model-usage' ? 120_000 : 60_000),
    summary: id === ids[0] ? marker : `${id} synthetic report`,
    proves: 'Only this local synthetic assertion.', doesNotProve: 'No deployment, provider or Production claim.',
  }))
  items[0].checks = [{
    id: health ? 'app-prozess' : 'domain-flights', name: health ? 'App-Prozess' : 'Flights',
    status: health ? 'healthy' : 'available', source: 'local-independent-get401-acceptance', freshness: fresh(60_000),
    summary: 'Synthetic child', proves: 'Only this child.', doesNotProve: 'No broader health claim.',
  }]
  return { checkedAt: iso(checkedAt), writeActions: [], items }
}

async function snapshot(page, markers) {
  return page.evaluate(markers => {
    const section = document.querySelector('[data-operations-phase]')
    const text = document.body.textContent ?? ''
    return {
      phase: section?.getAttribute('data-operations-phase'), pending: section?.getAttribute('data-operations-pending'),
      ariaBusy: section?.getAttribute('aria-busy'), refreshDisabled: document.querySelector('button')?.disabled,
      cards: Array.from(document.querySelectorAll('[data-health-id],[data-ops-id]')).map(node => ({
        id: node.getAttribute('data-health-id') ?? node.getAttribute('data-ops-id'),
        checkedAt: node.getAttribute('data-health-checked-at') ?? node.getAttribute('data-ops-checked-at'),
      })),
      alerts: Array.from(document.querySelectorAll('[role="alert"]')).map(node => node.textContent),
      markers: Object.fromEntries(Object.entries(markers).map(([key, value]) => [key, text.includes(value)])),
      historicalNotice: text.includes('Die angezeigte Prüfung ist historisch und unbestätigt.'),
      fetchCalls: window.__calls.map(call => ({ id: call.id, url: call.url, method: call.method, cache: call.cache })),
      responses: window.__responses.map(entry => ({ ...entry })),
      unresolvedRequestIds: window.__requests.filter(entry => !entry.resolved).map(entry => entry.id),
    }
  }, markers)
}
async function createPage(board, marker) {
  const context = await browser.newContext({ viewport: result.viewport, reducedMotion: result.reducedMotion })
  await context.clock.install({ time: base })
  await context.route('**/*', route => {
    const url = route.request().url()
    if (new URL(url).origin !== origin) { result.externalRequestAttempts.push(url); return route.abort() }
    result.localRequests.push(new URL(url).pathname); return route.continue()
  })
  await context.addInitScript(({ board, initial, source }) => {
    window.__FALL__ = { board, anfang: initial, endpunkt: source, erlaubt: true, sichtbar: true }
    window.__calls = []; window.__requests = []; window.__responses = []
    window.fetch = (url, init) => {
      const id = window.__calls.length + 1
      window.__calls.push({ id, url: String(url), method: String(init?.method ?? 'GET').toUpperCase(), cache: init?.cache })
      return new Promise((resolve, reject) => window.__requests.push({ id, resolved: false, resolve, reject }))
    }
  }, { board, initial: report(board, marker), source: endpoint(board) })
  const page = await context.newPage()
  page.setDefaultTimeout(8000)
  page.on('pageerror', error => result.pageErrors.push({ board, error: String(error) }))
  await page.goto(origin, { waitUntil: 'load' })
  await page.waitForFunction(marker => document.querySelector('[data-operations-phase]')?.getAttribute('data-operations-phase') === 'aktuell' && document.body.textContent.includes(marker), marker)
  return { page, context }
}
async function startRead(page, board, id, markers) {
  assert.equal(await page.getByRole('button', { name: 'Erneut prüfen', exact: true }).isDisabled(), false)
  await page.getByRole('button', { name: 'Erneut prüfen', exact: true }).click()
  await page.waitForFunction(id => window.__calls.length === id && document.querySelector('[data-operations-pending]')?.getAttribute('data-operations-pending') === 'true', id)
  const pending = await snapshot(page, markers)
  assert.equal(pending.pending, 'true')
  assert.equal(pending.ariaBusy, 'true')
  assert.equal(pending.refreshDisabled, true)
  assert.deepEqual(pending.unresolvedRequestIds, [id])
  assert.deepEqual(pending.fetchCalls.at(-1), { id, url: endpoint(board), method: 'GET', cache: 'no-store' })
  return pending
}
async function respond(page, id, spec) {
  await page.evaluate(({ id, spec }) => {
    const request = window.__requests.find(entry => entry.id === id && !entry.resolved)
    if (!request) throw new Error(`No unresolved local request ${id}`)
    request.resolved = true
    const record = { id, status: spec.status, malformed: !!spec.malformed, jsonAttempted: false, jsonReturned: false, jsonThrew: false }
    window.__responses.push(record)
    request.resolve({
      status: spec.status, ok: spec.status >= 200 && spec.status < 300,
      json: async () => {
        record.jsonAttempted = true
        if (spec.malformed) { record.jsonThrew = true; throw new SyntaxError('Local malformed JSON fixture') }
        record.jsonReturned = true
        return spec.body
      },
    })
  }, { id, spec })
}
async function waitSettled(page, phase) {
  await page.waitForFunction(phase => {
    const section = document.querySelector('[data-operations-phase]')
    return section?.getAttribute('data-operations-phase') === phase && section?.getAttribute('data-operations-pending') === 'false' && document.querySelector('button')?.disabled === false
  }, phase)
  await settle(page)
}
async function runCase(board, spec) {
  let page
  let context
  const markers = { baseline: `BASELINE-${board}-${spec.name}`, malformed: `INVALID-MUST-NOT-RENDER-${board}-${spec.name}`, recovery: `RECOVERED-${board}-${spec.name}` }
  try {
    ({ page, context } = await createPage(board, markers.baseline))
    const initial = await snapshot(page, markers)
    assert.equal(initial.phase, 'aktuell')
    assert.equal(initial.pending, 'false')
    assert.equal(initial.cards.length, expectedCards(board))
    assert.equal(initial.markers.baseline, true)
    assert.equal(initial.fetchCalls.length, 0)
    assert.deepEqual(initial.alerts, [])
    const pending = await startRead(page, board, 1, markers)
    const body = spec.mutate ? spec.mutate(report(board, markers.malformed)) : spec.body
    await respond(page, 1, { status: spec.status, body, malformed: spec.malformed })
    await waitSettled(page, spec.status === 200 ? 'historisch' : 'verweigert')
    const settled = await snapshot(page, markers)
    assert.equal(settled.pending, 'false')
    assert.equal(settled.ariaBusy, 'false')
    assert.equal(settled.refreshDisabled, false)
    assert.equal(settled.fetchCalls.length, 1)
    assert.deepEqual(settled.unresolvedRequestIds, [])
    assert.equal(settled.responses.length, 1)
    assert.equal(settled.responses[0].id, 1)
    assert.equal(settled.responses[0].jsonAttempted, true)
    assert.equal(settled.markers.malformed, false)
    if (spec.status === 200) {
      assert.equal(settled.responses[0].jsonReturned, true)
      assert.equal(settled.responses[0].jsonThrew, false)
      assert.equal(settled.phase, 'historisch')
      assert.deepEqual(settled.cards, initial.cards)
      assert.equal(settled.markers.baseline, true)
      assert.equal(settled.historicalNotice, true)
      assert.deepEqual(settled.alerts, [unreadable])
    } else {
      assert.equal(settled.phase, 'verweigert')
      assert.equal(settled.cards.length, 0)
      assert.equal(settled.markers.baseline, false)
      assert.equal(settled.historicalNotice, false)
      assert.deepEqual(settled.alerts, [spec.expectedAlert])
      assert.equal(settled.responses[0].jsonThrew, !!spec.malformed)
    }
    const recoveryPending = await startRead(page, board, 2, markers)
    await respond(page, 2, { status: 200, body: report(board, markers.recovery, await page.evaluate(() => Date.now())) })
    await waitSettled(page, 'aktuell')
    const recovered = await snapshot(page, markers)
    assert.equal(recovered.markers.recovery, true)
    assert.equal(recovered.markers.baseline, false)
    assert.equal(recovered.markers.malformed, false)
    assert.equal(recovered.cards.length, expectedCards(board))
    assert.equal(recovered.historicalNotice, false)
    assert.deepEqual(recovered.alerts, [])
    assert.equal(recovered.fetchCalls.length, 2)
    assert.deepEqual(recovered.unresolvedRequestIds, [])
    assert.equal(recovered.responses[1].id, 2)
    assert.equal(recovered.responses[1].jsonReturned, true)
    result.cases.push({ board, name: spec.name, status: spec.status, accepted: true, requestBodySha256: spec.malformed ? null : createHash('sha256').update(JSON.stringify(body)).digest('hex'), initial, pending, settled, recoveryPending, recovered })
    console.log(`GET401_CASE_ACCEPTED ${board} ${spec.name}`)
  } catch (error) {
    result.cases.push({ board, name: spec.name, status: spec.status, accepted: false, failure: String(error?.stack ?? error), observed: page ? await snapshot(page, markers).catch(() => null) : null })
    console.error(`GET401_CASE_FAILED ${board} ${spec.name}: ${String(error)}`)
  } finally { await context?.close() }
}
