function sourceSnapshot() {
  assert.equal(git('rev-parse', 'HEAD'), expectedHead, 'HEAD changed during acceptance.')
  assert.equal(git('status', '--porcelain'), '', 'Checkout became dirty during acceptance.')
  return Object.fromEntries(sourcePaths.map(path => {
    const bytes = readFileSync(join(repo, path))
    const expectedGitBlob = git('rev-parse', `${expectedHead}:${path}`)
    const actualGitBlob = createHash('sha1').update(`blob ${bytes.length}\0`).update(bytes).digest('hex')
    assert.equal(actualGitBlob, expectedGitBlob, `Source differs from explicit committed head: ${path}`)
    return [path, { gitBlob: actualGitBlob, sha256: createHash('sha256').update(bytes).digest('hex') }]
  }))
}

function report(board, marker = 'INITIAL-ACCEPTED-REPORT', checkedAt = base) {
  const health = board === 'health'
  const ids = health ? ['app', 'vercel', 'supabase', 'github', 'infomaniak'] : ['provider-ops', 'kill-switch', 'cost-guard', 'model-usage']
  const fresh = ttlMs => ({ state: 'fresh', ageMs: 0, ttlMs })
  const items = ids.map(id => ({
    id, name: id,
    status: health ? (id === 'app' ? 'unknown' : 'not_configured') : (id === 'model-usage' ? 'empty' : 'foundation_only'),
    source: 'local-independent-acceptance', checkedAt: iso(checkedAt),
    freshness: fresh(id === 'github' || id === 'infomaniak' ? 300_000 : id === 'vercel' || id === 'model-usage' ? 120_000 : 60_000),
    summary: id === ids[0] ? marker : `${id} synthetic report`,
    proves: 'Only this local synthetic assertion.', doesNotProve: 'No deployment, provider integration or Production.',
  }))
  items[0].checks = [{
    id: health ? 'app-prozess' : 'domain-flights', name: health ? 'App-Prozess' : 'Flights',
    status: health ? 'healthy' : 'available', source: 'local-independent-acceptance', freshness: fresh(60_000),
    summary: 'Synthetic child', proves: 'Only the synthetic child.', doesNotProve: 'No broader health claim.',
  }]
  return { checkedAt: iso(checkedAt), writeActions: [], items }
}

async function snapshot(page, board) {
  return page.evaluate(selectors => ({
    phase: document.querySelector('[data-operations-phase]')?.getAttribute('data-operations-phase'),
    pending: document.querySelector('[data-operations-pending]')?.getAttribute('data-operations-pending'),
    refreshDisabled: document.querySelector('button')?.disabled,
    cardFreshness: document.querySelector(selectors.card)?.getAttribute(selectors.freshness),
    otherFreshness: document.querySelector(selectors.other)?.getAttribute(selectors.freshness),
    childFreshness: document.querySelector(selectors.child)?.getAttribute(selectors.freshness),
    childGreen: document.querySelector(selectors.child)?.getAttribute(selectors.green),
    cardCheckedAt: document.querySelector(selectors.card)?.getAttribute(selectors.checkedAt),
    cardCount: document.querySelectorAll('[data-health-id], [data-ops-id]').length,
    wall: Date.now(), monotonicFinite: Number.isFinite(performance.now()),
    fetchCalls: window.__calls.map(call => ({ url: call.url, cache: call.cache, signalAborted: call.signal?.aborted ?? null })),
    responsesResolved: window.__resolved, responseBodiesRead: window.__bodiesRead,
    propsWaiters: window.__propsWaiters.length, text: document.body.textContent,
  }), selected(board))
}
async function createPage(board, { held = false, initial = report(board) } = {}) {
  const context = await browser.newContext({ viewport: result.viewport, reducedMotion: result.reducedMotion })
  await context.clock.install({ time: base })
  await context.route('**/*', route => {
    const url = route.request().url()
    if (new URL(url).origin !== origin) { result.externalRequestAttempts.push(url); return route.abort() }
    result.localRequests.push(new URL(url).pathname)
    return route.continue()
  })
  await context.addInitScript(({ board, initial, held, source }) => {
    window.__FALL__ = { board, anfang: initial, endpunkt: source, erlaubt: true, sichtbar: true }
    window.__calls = []; window.__requests = []; window.__propsWaiters = []
    window.__resolved = 0; window.__bodiesRead = 0
    window.fetch = (url, init) => {
      window.__calls.push({ url: String(url), cache: init?.cache, signal: init?.signal })
      return new Promise((resolve, reject) => window.__requests.push({ resolve, reject }))
    }
    if (held) window.__JETNITY_OPS_PROPS_SPERRE__ = () => new Promise(resolve => window.__propsWaiters.push(resolve))
  }, { board, initial, held, source: endpoint(board) })
  const page = await context.newPage()
  page.setDefaultTimeout(8000)
  page.on('pageerror', error => result.pageErrors.push({ board, error: String(error) }))
  await page.goto(origin, { waitUntil: 'load' })
  await page.waitForFunction(() => typeof window.__setze === 'function')
  return { page, context }
}
async function resolveGet(page, body) {
  await page.evaluate(body => {
    const request = window.__requests.shift()
    if (!request) throw new Error('Expected one pending local fetch')
    window.__resolved++
    request.resolve({ ok: true, status: 200, json: async () => { window.__bodiesRead++; return body } })
  }, body)
  // Correct implementations may discard a stale response before reading its body.
  await settle(page)
}
async function explicitCurrentRead(page, board, marker, expectedCalls) {
  const button = page.getByRole('button', { name: 'Erneut prüfen', exact: true })
  assert.equal(await button.isDisabled(), false)
  await button.click()
  await page.waitForFunction(count => window.__calls.length === count, expectedCalls)
  const pending = await snapshot(page, board)
  assert.equal(pending.pending, 'true')
  assert.equal(pending.fetchCalls.length, expectedCalls)
  assert.equal(pending.fetchCalls.at(-1).url, endpoint(board))
  assert.equal(pending.fetchCalls.at(-1).cache, 'no-store')
  await resolveGet(page, report(board, marker, await page.evaluate(() => Date.now())))
  await waitMarker(page, marker)
  const accepted = await snapshot(page, board)
  assert.equal(accepted.phase, 'aktuell')
  assert.equal(accepted.pending, 'false')
  assert.equal(accepted.refreshDisabled, false)
  return accepted
}
async function caseFor(name, board, options, fn) {
  let context
  let page
  try {
    ({ page, context } = await createPage(board, options))
    const observations = await fn(page)
    result.cases.push({ name, board, accepted: true, observations })
    console.log(`FOCUSED_CASE_ACCEPTED ${board} ${name}`)
  } catch (error) {
    const observed = page ? await snapshot(page, board).catch(() => null) : null
    result.cases.push({ name, board, accepted: false, observed, failure: String(error?.stack ?? error) })
    console.error(`FOCUSED_CASE_FAILED ${board} ${name}: ${String(error)}`)
  } finally { await context?.close() }
}
