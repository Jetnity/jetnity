import assert from 'node:assert/strict'
import { readFileSync, realpathSync } from 'node:fs'
import { join } from 'node:path'
import { pathToFileURL } from 'node:url'
import { argumentsForRun, moduleApi, prepareRun, sha256, writeReport } from './trip-acceptance-common.mjs'

// Revalidate the fixture/builder exports and mounted Host API at the new exact head before execution.
const REJECTED_HEADS = new Set([
  '80006fdf1443ca9d75d8c481246821354da16000',
  'b218c33d9d00c19f3705aba8bd208cc455272659',
])

async function run() {
  const options = argumentsForRun()
  const report: any = {
    purpose: 'Independent correct-behavior assertions for R1 same-ref locking and newer B-control focus; not a TL PASS',
    expectedHead: options.head, node: process.version, generatedAt: new Date().toISOString(),
    observations: [], verdict: 'NOT_VERIFIED',
  }
  let browser: any
  try {
    assert.ok(!REJECTED_HEADS.has(options.head), 'R2 acceptance requires an explicit new candidate SHA; refusing either demonstrated defective head')
    const context = prepareRun(options, [
      'components/trips/EigeneReisevorbereitung.tsx', 'components/trips/usePersonalPreparationLifecycle.ts',
      'lib/readiness/personal-preparation-workflow.ts', 'lib/readiness/bauen.ts',
      'lib/reiseaenderung/fixtures/reise.ts', 'scripts/trip-personal-preparation-workflow-completion-1/flaeche.tsx',
    ])
    assert.ok(options.bundle && options['bundle-info'], 'Use the bundle and metadata from the guarded builder')
    const bundleInfo = JSON.parse(readFileSync(options['bundle-info'], 'utf8'))
    assert.equal(bundleInfo.head, context.head)
    assert.equal(realpathSync(bundleInfo.repo), context.repo)
    assert.equal(realpathSync(bundleInfo.bundle), realpathSync(options.bundle))
    assert.equal(sha256(options.bundle), bundleInfo.bundleSha256)
    for (const [path, expected] of Object.entries(bundleInfo.sourceBlobs)) assert.equal(context.verifySource(path), expected)
    const workflow = moduleApi(await import(pathToFileURL(join(context.repo, 'lib/readiness/personal-preparation-workflow.ts')).href))
    const fixture = moduleApi(await import(pathToFileURL(join(context.repo, 'lib/reiseaenderung/fixtures/reise.ts')).href))
    const builder = moduleApi(await import(pathToFileURL(join(context.repo, 'lib/readiness/bauen.ts')).href))
    const playwright = await import(pathToFileURL(join(context.repo, 'node_modules/playwright/index.mjs')).href)
    browser = await playwright.chromium.launch({ headless: true })
    const bundle = readFileSync(options.bundle, 'utf8')
    const empty = fixture.beispielreise({ id: 'tl-r1-mounted-order-trip', readinessItems: [] })
    const built = builder.readinessItemBauen(empty, {
      clientRef: 'preparation-tl-existing-b', kind: 'preparation', userStatus: 'open',
      countryCode: null, tripItemId: null, title: 'Existing B',
    })
    assert.ok(built.ok)
    const basis = { ...empty, readinessItems: [built.item] }
    let externalRequests = 0
    async function frame(page: any) {
      await page.evaluate(() => new Promise<void>((resolve) => requestAnimationFrame(() => requestAnimationFrame(() => resolve()))))
    }
    async function open(seam: 'account' | 'guest') {
      const page = await browser.newPage({ viewport: { width: 390, height: 844 } })
      page.setDefaultTimeout(5000)
      await page.route('**/*', async (route: any) => { externalRequests += 1; await route.abort() })
      const start = { seam, reise: basis, ...(seam === 'account' ? { kanonisch: basis } : {}) }
      await page.setContent(`<!doctype html><html><body><div id="root"></div><script>window.__START__=${JSON.stringify(start)}</script><script>${bundle}</script></body></html>`)
      await page.waitForFunction(() => (window as any).__PERSONAL__, undefined, { timeout: 5000 })
      const keys = await page.evaluate(() => Object.keys((window as any).__PERSONAL__))
      for (const name of ['setzen', 'entfernen', 'lesen', 'schreibenFreigeben', 'lesenFreigeben', 'reiseSetzen', 'kanonischSetzen']) assert.ok(keys.includes(name), `Inspected Host API missing ${name}`)
      return page
    }
    async function check(name: string, seam: 'account' | 'guest', fn: (page: any, observation: any) => Promise<void>) {
      let page: any
      const observation: any = { name, result: 'NOT_VERIFIED' }
      report.observations.push(observation)
      try { page = await open(seam); await fn(page, observation); observation.result = 'PASS' }
      catch (error) { observation.result = 'FAIL'; observation.error = String(error) }
      finally { await page?.close() }
    }
    await check('Old A conflict read must not allow ordinary delete during newer same-ref B write', 'account', async (page, observation) => {
      const row = page.locator(`[data-personal-client-ref="${built.item.clientRef}"]`)
      await row.getByRole('button', { name: 'Titel bearbeiten', exact: true }).click()
      const title = row.getByLabel('Titel bearbeiten', { exact: true })
      await title.fill('First accepted edit')
      await row.getByRole('button', { name: 'Speichern', exact: true }).click()
      await page.waitForFunction(() => (window as any).__PERSONAL__.setzen.length === 1)
      await page.evaluate(() => (window as any).__PERSONAL__.schreibenFreigeben(null))
      await page.waitForFunction(() => (window as any).__PERSONAL__.lesen === 1)
      await page.evaluate(() => (window as any).__PERSONAL__.lesenFreigeben({ art: 'props' }))
      await page.evaluate((trip: any) => (window as any).__PERSONAL__.kanonischSetzen(trip), { ...basis })
      await page.waitForFunction((ref: string) => document.querySelector(`[data-personal-client-ref="${ref}"]`)?.getAttribute('data-personal-phase') === 'conflict', built.item.clientRef)
      await title.fill('Second pending edit')
      await row.getByRole('button', { name: 'Speichern', exact: true }).click()
      await page.waitForFunction(() => (window as any).__PERSONAL__.setzen.length === 2)
      assert.equal(await row.getAttribute('data-personal-phase'), 'pending')
      await page.evaluate((trip: any) => (window as any).__PERSONAL__.kanonischSetzen(trip), { ...basis })
      await page.waitForFunction(() => Number(document.querySelector('[data-personal-hook-lieferung]')?.getAttribute('data-personal-hook-lieferung')) >= 3)
      await frame(page)
      const phaseBeforeDelete = await row.getAttribute('data-personal-phase')
      await row.getByRole('button', { name: 'Entfernen', exact: true }).click()
      const confirm = row.getByRole('button', { name: 'Endgültig entfernen', exact: true })
      const enabled = await confirm.isEnabled()
      if (enabled) await confirm.click()
      await frame(page)
      observation.detail = {
        phaseBeforeDelete,
        deleteButtonEnabled: enabled,
        observed: await page.evaluate(() => ({
          writes: (window as any).__PERSONAL__.setzen,
          deletes: (window as any).__PERSONAL__.entfernen,
          reads: (window as any).__PERSONAL__.lesen,
        })),
        secondWriteReleased: false,
      }
      assert.equal(phaseBeforeDelete, 'pending', 'Newer B must still be outstanding for this ordering probe')
      assert.equal(observation.detail.observed.deletes.length, 0, 'Same-ref delete must not dispatch while newer B is still pending')
    })
    for (const target of ['status', 'delete-confirmation']) {
      await check(`Late A completion must preserve deliberate B ${target} focus`, 'guest', async (page, observation) => {
        const rowB = page.locator(`[data-personal-client-ref="${built.item.clientRef}"]`)
        if (target === 'status') await rowB.getByRole('button', { name: 'Titel bearbeiten', exact: true }).click()
        await page.getByLabel('Eigene Vorbereitung', { exact: true }).fill('Pending A')
        await page.getByRole('button', { name: 'Punkt hinzufügen', exact: true }).click()
        await page.waitForFunction(() => (window as any).__PERSONAL__.setzen.length === 1)
        const payload = await page.evaluate(() => (window as any).__PERSONAL__.setzen[0])
        const a = builder.readinessItemBauen(basis, payload)
        assert.ok(a.ok)
        const matching = { ...basis, readinessItems: [...basis.readinessItems, a.item] }
        await page.evaluate((trip: any) => (window as any).__PERSONAL__.reiseSetzen(trip), matching)
        await page.evaluate(() => (window as any).__PERSONAL__.schreibenFreigeben(null))
        await page.waitForFunction(() => (window as any).__PERSONAL__.lesen === 1)
        let targetButton: any
        if (target === 'status') {
          targetButton = rowB.getByRole('button', { name: 'Erledigt', exact: true })
          await targetButton.click()
          assert.equal(await targetButton.getAttribute('aria-pressed'), 'true')
        } else {
          await rowB.getByRole('button', { name: 'Entfernen', exact: true }).click()
          targetButton = rowB.getByRole('button', { name: 'Endgültig entfernen', exact: true })
          await targetButton.focus()
        }
        assert.equal(await targetButton.evaluate((element: Element) => document.activeElement === element), true)
        await page.evaluate((trip: any) => (window as any).__PERSONAL__.lesenFreigeben({ art: 'success', reise: trip }), matching)
        await page.waitForFunction((ref: string) => ['confirmed', 'reconciled'].includes(document.querySelector(`[data-personal-client-ref="${ref}"]`)?.getAttribute('data-personal-phase') ?? ''), a.item.clientRef)
        await frame(page)
        const retained = await targetButton.evaluate((element: Element) => document.activeElement === element)
        observation.detail = {
          intendedTarget: `B ${target}`,
          retained,
          active: await page.evaluate(() => ({ tag: document.activeElement?.tagName, text: document.activeElement?.textContent, ref: document.activeElement?.getAttribute('data-personal-client-ref') })),
          statusSelectionRetained: target === 'status' ? await targetButton.getAttribute('aria-pressed') : null,
          counts: await page.evaluate(() => ({ writes: (window as any).__PERSONAL__.setzen.length, deletes: (window as any).__PERSONAL__.entfernen.length, reads: (window as any).__PERSONAL__.lesen })),
        }
        assert.equal(retained, true, 'Late A completion must not steal focus from a deliberately used control of B')
      })
    }
    assert.equal(externalRequests, 0)
    report.head = context.head
    report.browser = browser.version()
    report.externalRequests = externalRequests
    report.verifiedSources = context.verifiedSources
    report.bundleSha256 = bundleInfo.bundleSha256
    report.scope = 'Actual production list and lifecycle hook in the unchanged inspected synthetic flaeche Host. Deferred adapters record dispatches. No real storage CRUD, full TripWorkspace, Next router, hosted Account, Supabase/RLS or Production claim.'
    report.verdict = report.observations.every((entry: any) => entry.result === 'PASS') ? 'BOUNDED_ASSERTIONS_PASSED' : 'BOUNDED_ASSERTION_FAILED'
  } catch (error) { report.error = String(error) }
  finally { await browser?.close() }
  writeReport(options, report)
  if (report.verdict !== 'BOUNDED_ASSERTIONS_PASSED') process.exitCode = 1
}

run().catch((error) => { console.error(error); process.exitCode = 1 })
