const FILTER: BuchungenFilter = { titel: 'Zürich', art: 'flight', archiv: 'aktiv' }

function neueAusgabe(): string {
  const index = process.argv.indexOf('--ausgabe')
  const ziel = index >= 0 ? process.argv[index + 1] : ''
  if (!ziel) throw new Error('Aufruf braucht --ausgabe <neues Verzeichnis>')
  if (existsSync(ziel)) throw new Error(`Ausgabe existiert bereits und wird nicht überschrieben: ${ziel}`)
  mkdirSync(ziel, { recursive: true })
  return ziel
}

function git(args: string[]): string {
  const ergebnis = spawnSync('git', args, { encoding: 'utf8' })
  if (ergebnis.status !== 0) throw new Error(ergebnis.stderr || ergebnis.stdout)
  return ergebnis.stdout.trim()
}

function uuid(n: number): string {
  return `00000000-0000-4000-8000-${n.toString(16).padStart(12, '0')}`
}

function zeile(n: number): RoheBuchungszeile & { booking_confirmed_at: string } {
  return {
    id: uuid(n),
    kind: 'flight',
    title: n === 1 ? `${'Langtitel'.repeat(12)} Zürich` : `Flug ${n}`,
    starts_on: '2026-09-12',
    starts_at: '08:40:00',
    ends_on: '2026-09-12',
    ends_at: null,
    booking_status: 'booked',
    booking_confirmed_at: '2026-08-30T12:00:00.123456Z',
    trips: { id: uuid(90), title: 'Lissabon', status: n % 2 === 0 ? 'archived' : 'planned' },
  }
}

function ansicht(zustand: 'liste' | 'fortsetzung' | 'leer' | 'keine' | 'fehler' | 'ungueltig' | 'fortsetzungLeer') {
  if (zustand === 'liste') {
    const seite = buchungenSeiteAusZeilen(
      Array.from({ length: 51 }, (_, index) => zeile(index + 1)),
      FILTER,
    )
    if (!seite.ok) throw new Error('Liste konnte nicht abgebildet werden')
    return buchungenAnsichtAus({
      ok: true,
      filter: FILTER,
      fortsetzung: false,
      buchungen: seite.buchungen,
      weiter: `/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv&weiter=${seite.weiter}`,
      ersteSeite: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
      wiederholen: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
    })
  }
  if (zustand === 'fortsetzung') {
    const seite = buchungenSeiteAusZeilen([zeile(1), zeile(2)], FILTER)
    if (!seite.ok) throw new Error('Fortsetzung konnte nicht abgebildet werden')
    return buchungenAnsichtAus({
      ok: true,
      filter: FILTER,
      fortsetzung: true,
      buchungen: seite.buchungen,
      weiter: null,
      ersteSeite: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
      wiederholen: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv&weiter=cursor',
    })
  }
  if (zustand === 'leer') {
    return buchungenAnsichtAus({
      ok: true,
      filter: { titel: null, art: null, archiv: 'alle' },
      fortsetzung: false,
      buchungen: [],
      weiter: null,
      ersteSeite: '/account/bookings',
      wiederholen: '/account/bookings',
    })
  }
  if (zustand === 'keine') {
    return buchungenAnsichtAus({
      ok: true,
      filter: FILTER,
      fortsetzung: false,
      buchungen: [],
      weiter: null,
      ersteSeite: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
      wiederholen: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
    })
  }
  if (zustand === 'fehler') {
    return buchungenAnsichtAus({
      ok: false,
      grund: 'lesen',
      status: 500,
      filter: FILTER,
      ersteSeite: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
      wiederholen: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
    })
  }
  if (zustand === 'fortsetzungLeer') {
    return buchungenAnsichtAus({
      ok: true,
      filter: FILTER,
      fortsetzung: true,
      buchungen: [],
      weiter: null,
      ersteSeite: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
      wiederholen: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv&weiter=cursor',
    })
  }
  return buchungenAnsichtAus({
    ok: false,
    grund: 'cursor',
    filter: FILTER,
    ersteSeite: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv',
    wiederholen: '/account/bookings?titel=Z%C3%BCrich&art=flight&archiv=aktiv&weiter=kaputt',
  })
}

function cssDateien(ordner: string, treffer: string[] = []): string[] {
  for (const eintrag of readdirSync(ordner, { withFileTypes: true })) {
    const pfad = join(ordner, eintrag.name)
    if (eintrag.isDirectory()) cssDateien(pfad, treffer)
    else if (eintrag.name.endsWith('.css')) treffer.push(pfad)
  }
  return treffer
}

function sha256(inhalt: Buffer | string): string {
  return createHash('sha256').update(inhalt).digest('hex')
}

function cssLesen(): { text: string; sha256: string; dateien: { pfad: string; sha256: string; bytes: number }[] } {
  const pfad = process.env.JETNITY_REVIEW_CSS_PATH
  if (!pfad) throw new Error('Verifizierte lokale Projekt-PostCSS-Datei fehlt.')
  const inhalt = readFileSync(pfad)
  const text = inhalt.toString('utf8')
  return {
    text,
    sha256: sha256(text),
    dateien: [{ pfad, sha256: sha256(inhalt), bytes: inhalt.length }],
  }
}

type Gemessen = {
  clientWidth: number
  scrollWidth: number
  ueberlauf: number
  schrift: string
  zustand: string | null
  h1: {
    fontSize: string
    lineHeight: string
    letterSpacing: string
    fontWeight: string
    overflowWrap: string
  } | null
  link: {
    text: string
    fontSize: string
    backgroundColor: string
    display: string
    overflowWrap: string
    rechts: number
    breite: number
  } | null
}

function zahl(wert: string): number {
  const n = Number.parseFloat(wert)
  if (!Number.isFinite(n)) throw new Error(`Keine Zahl: ${wert}`)
  return n
}

function stilePruefen(breite: number, schrift: string, mass: Gemessen): void {
  if (mass.schrift !== schrift) throw new Error(`Root-Schrift ${mass.schrift} statt ${schrift}`)
  if (!mass.h1) throw new Error('H1 fehlt. Die Überschrift wurde nicht gerendert.')
  const wurzel = zahl(schrift)
  const h1 = zahl(mass.h1.fontSize)
  const erwartet = (breite >= 640 ? 3 : 1.875) * wurzel
  if (Math.abs(h1 - erwartet) > 0.6) {
    throw new Error(
      `H1 ${mass.h1.fontSize} statt ${erwartet}px bei Root ${schrift} und Breite ${breite}. text-3xl/text-5xl ist nicht wirksam.`,
    )
  }
  if (mass.h1.fontWeight !== '600') throw new Error(`H1-Gewicht ${mass.h1.fontWeight} statt 600`)
  if (mass.h1.overflowWrap !== 'anywhere') throw new Error(`H1 overflow-wrap ${mass.h1.overflowWrap}`)
  const abstand = zahl(mass.h1.letterSpacing)
  if (Math.abs(abstand - -0.04 * h1) > 0.25) {
    throw new Error(`H1-Laufweite ${mass.h1.letterSpacing} statt ${-0.04 * h1}px`)
  }
  if (schrift === '32px' && breite === 320) {
    const zeile = zahl(mass.h1.lineHeight)
    if (Math.abs(zeile - 72) > 0.6) throw new Error(`H1-Zeilenhöhe ${mass.h1.lineHeight} statt 72px`)
  }
  if (!mass.link) return
  if (mass.link.backgroundColor === 'rgba(0, 0, 0, 0)' || mass.link.backgroundColor === 'transparent') {
    throw new Error('Aktionslink ohne Hintergrund. Production-CSS ist nicht wirksam.')
  }
  if (mass.link.display !== 'inline-flex' && mass.link.display !== 'flex') {
    throw new Error(`Link-Display ${mass.link.display} bei „${mass.link.text}“`)
  }
  if (mass.link.overflowWrap !== 'anywhere') throw new Error(`Link overflow-wrap ${mass.link.overflowWrap}`)
  const linkSchrift = zahl(mass.link.fontSize)
  if (Math.abs(linkSchrift - 0.875 * wurzel) > 0.6) {
    throw new Error(`Link-Schrift ${mass.link.fontSize} statt ${0.875 * wurzel}px`)
  }
  if (mass.link.rechts > mass.clientWidth + 1) {
    throw new Error(`Link ragt über den Viewport: ${mass.link.rechts} > ${mass.clientWidth}`)
  }
}

function dokument(markup: string, css: string): string {
  return `<!doctype html><html lang="de"><head><meta charset="utf-8"><style>${css}</style></head><body>${markup}</body></html>`
}

async function tabWeg(
  seite: import('playwright').Page,
  ziele: string[],
  grenze: number,
): Promise<string[]> {
  await seite.emulateMedia({ reducedMotion: 'reduce' })
  await seite.locator('input[name="titel"]').focus()
  const gesehen: string[] = []
  const offen = new Set(ziele)
  for (let schritt = 0; schritt < grenze && offen.size > 0; schritt += 1) {
    const name = await seite.evaluate(() => {
      const el = document.activeElement
      if (!el || el === document.body) return 'BODY'
      const text = (el.textContent ?? '').replace(/\s+/g, ' ').trim()
      return `${el.tagName}:${el.getAttribute('name') ?? text.slice(0, 80)}`
    })
    gesehen.push(name)
    for (const ziel of offen) {
      if (name === ziel || name.startsWith(`${ziel}`) || name.includes(ziel)) offen.delete(ziel)
    }
    await seite.keyboard.press('Tab')
  }
  if (offen.size > 0) {
    throw new Error(`Tastatur fehlt ${[...offen].join(', ')}: ${gesehen.filter((name, index, alle) => alle.indexOf(name) === index).join(' | ')}`)
  }
  return gesehen
}

async function montage(): Promise<void> {
const ausgabe = neueAusgabe()
const bilder = join(ausgabe, 'screens')
// Root-only passive observer; no routing, interception, fixture or navigation changes.
const rootDiagnostik = {
  pageerror: [] as string[],
  consoleError: [] as string[],
  unexpectedRequests: [] as string[],
  requestFailed: [] as string[],
}
mkdirSync(bilder, { recursive: true })
const css = cssLesen()
const quellenCss = ['styles/globals.css', 'tailwind.config.js', 'postcss.config.js'].map((pfad) => {
  const inhalt = readFileSync(pfad)
  return { pfad, sha256: sha256(inhalt), bytes: inhalt.length, git: git(['hash-object', pfad]) }
})
const browser = await chromium.launch({ executablePath: process.env.JETNITY_REVIEW_CHROMIUM })
const befund: {
  browser: string
  zustaende: Record<string, unknown>
  tastatur: Record<string, string[]>
  bewegung: string
  hinweis: string
} = {
  browser: browser.version(),
  zustaende: {},
  tastatur: {},
  bewegung: 'reduce',
  hinweis:
    'LOCAL_PROJECT_POSTCSS_COMPILATION: CSS aus der unveränderten Projekt-PostCSS-Pipeline, injiziert als style-Element. Kein Next-Build-CSS-Nachweis. Berechnete H1-, Link- und Root-Stile werden mitgeschrieben und gegen die Utilities geprüft. Kein physisches Gerät, kein nativer Zoom, kein Screenreader, kein gehostetes Konto. Die frühere Montage mit Überlauf 0 hat die berechnete H1-Schrift nicht festgehalten; sie wird hier nicht umgedeutet.',
}

try {
  for (const zustand of zustaende) {
    const markup = renderToStaticMarkup(createElement(AccountBuchungen, { ansicht: ansicht(zustand) }))
    const seite = await browser.newPage()
    seite.on('pageerror', (problem) => rootDiagnostik.pageerror.push(problem.message))
    seite.on('console', (meldung) => {
      if (meldung.type() === 'error') rootDiagnostik.consoleError.push(meldung.text())
    })
    seite.on('request', (anfrage) => {
      const url = anfrage.url()
      if (!url.startsWith('data:') && !url.startsWith('blob:') && url !== 'about:blank') {
        rootDiagnostik.unexpectedRequests.push(url)
      }
    })
    seite.on('requestfailed', (anfrage) => {
      rootDiagnostik.requestFailed.push(anfrage.url() + ' ' + (anfrage.failure()?.errorText ?? 'unknown'))
    })
    const messungen = []
    for (const breite of breiten) {
      await seite.setViewportSize({ width: breite.width, height: breite.height })
      await seite.setContent(dokument(markup, css.text), { waitUntil: 'domcontentloaded' })
      const schrift = breite.name === '320' ? '32px' : '16px'
      await seite.evaluate((groesse) => {
        document.documentElement.style.fontSize = groesse
      }, schrift)
      const mass = await seite.evaluate((): Gemessen => {
        const h1 = document.querySelector('h1')
        const h1Stil = h1 ? getComputedStyle(h1) : null
        const link = [...document.querySelectorAll('a')].find((anker) => (anker.textContent ?? '').includes('Planpunkt öffnen'))
          ?? [...document.querySelectorAll('a')].find((anker) => (anker.textContent ?? '').trim().length > 0)
        const linkStil = link ? getComputedStyle(link) : null
        const linkBereich = link?.getBoundingClientRect()
        return {
          clientWidth: document.documentElement.clientWidth,
          scrollWidth: document.documentElement.scrollWidth,
          ueberlauf: document.documentElement.scrollWidth - document.documentElement.clientWidth,
          schrift: getComputedStyle(document.documentElement).fontSize,
          zustand: document.querySelector('[data-buchungen]')?.getAttribute('data-buchungen') ?? null,
          h1: h1Stil
            ? {
                fontSize: h1Stil.fontSize,
                lineHeight: h1Stil.lineHeight,
                letterSpacing: h1Stil.letterSpacing,
                fontWeight: h1Stil.fontWeight,
                overflowWrap: h1Stil.overflowWrap,
              }
            : null,
          link: link && linkStil && linkBereich
            ? {
                text: (link.textContent ?? '').replace(/\s+/g, ' ').trim().slice(0, 80),
                fontSize: linkStil.fontSize,
                backgroundColor: linkStil.backgroundColor,
                display: linkStil.display,
                overflowWrap: linkStil.overflowWrap,
                rechts: linkBereich.right,
                breite: linkBereich.width,
              }
            : null,
        }
      })
      if (mass.ueberlauf > 1) throw new Error(`${zustand} ${breite.name} Überlauf ${mass.ueberlauf}`)
      stilePruefen(breite.width, schrift, mass)
      messungen.push({ ...breite, schrift, ueberlauf: mass.ueberlauf, zustand: mass.zustand, h1: mass.h1, link: mass.link })
    }
    await seite.setViewportSize({ width: 390, height: 844 })
    await seite.setContent(dokument(markup, css.text), { waitUntil: 'domcontentloaded' })
    const weg = tastaturZiele[zustand]
    const gesehen = await tabWeg(seite, weg.ziele, weg.grenze)
    befund.tastatur[zustand] = gesehen.filter((name, index, alle) => alle.indexOf(name) === index)
    if (zustand === 'leer') {
      const reset = await seite.getByRole('link', { name: 'Filter zurücksetzen' }).count()
      if (reset !== 0) throw new Error('Die leere erste Seite ohne Filter bietet Zurücksetzen an.')
    }
    if (zustand === 'liste' || zustand === 'leer' || zustand === 'keine' || zustand === 'fehler' || zustand === 'fortsetzungLeer') {
      await seite.setViewportSize({ width: 390, height: 844 })
      await seite.screenshot({
        path: join(bilder, `bookings-${zustand}-390.png`),
        fullPage: true,
      })
    }
    if (zustand === 'liste') {
      await seite.setViewportSize({ width: 320, height: 700 })
      await seite.evaluate(() => {
        document.documentElement.style.fontSize = '32px'
      })
      await seite.screenshot({ path: join(bilder, 'bookings-liste-320-32px.png'), fullPage: true })
    }
    await seite.close()
    befund.zustaende[zustand] = messungen
  }
  const quellen = [
    'scripts/account-booking-folder-completion-1/montage.ts',
    'components/account/AccountBuchungen.tsx',
    'components/account/BuchungenFilter.tsx',
    'lib/account/buchungen-abfrage.ts',
    'docs/JETNITY_ACCOUNT_BOOKING_FOLDER_COMPLETION_1_TASK_V1_2026-10-24.md',
    'docs/JETNITY_ACCOUNT_BOOKING_FOLDER_COMPLETION_1_PLAN_2026-10-24.md',
  ]
  const hashes: Record<string, string> = {}
  for (const pfad of quellen) hashes[pfad] = git(['hash-object', pfad])
  const bericht = {
    herkunft: {
      autor: 'Jetnity Account Booking Folder Completion 1 — Cursor Generation 1',
      session: 'bc-a34030c7-4c19-4c9f-863f-193cfdfa34da',
      modell: 'grok-4.7-high-fast',
      kopf: git(['rev-parse', 'HEAD']),
      baum: git(['rev-parse', 'HEAD^{tree}']),
      main: git(['rev-parse', '9092b29c2e187bd73d644a3d85670f3ac6436be1']),
      quellen: hashes,
      css: {
        herkunft: 'LOCAL_PROJECT_POSTCSS_COMPILATION: Eine verifizierte CSS-Datei aus der unveränderten Projekt-PostCSS-Pipeline. Kein Next-Production-CSS-Gleichheitsnachweis; sha256/dateien dokumentieren genau diese Datei.',
        sha256: css.sha256,
        dateien: css.dateien,
        quellen: quellenCss,
      },
      befehl: process.argv.join(' '),
      node: process.version,
    },
    grenze:
      'Statisches Markup mit LOCAL_PROJECT_POSTCSS_COMPILATION aus der unveränderten Projekt-PostCSS-Pipeline. Kein Next-Build-CSS-Nachweis. Die berechnete H1-Schrift muss text-3xl (60px bei Root 32px, 30px bei Root 16px) oder ab 640px text-5xl sein; sonst bricht die Montage ab. Kein Hydration, kein App-Router, kein nativer Zoom, kein Screenreader, kein gehostetes Konto. Root-Font 32px nur bei Breite 320, sonst 16px. Die Montage von 1574adcb mit Überlauf 0 bleibt unverändert und ist kein Beleg für diese Schrift.',
    befund,
  }
  writeFileSync(join(ausgabe, 'montage.json'), `${JSON.stringify(bericht, null, 2)}\n`)
  console.log(JSON.stringify({ ausgabe, browser: befund.browser, tastatur: befund.tastatur, zustaende: Object.keys(befund.zustaende) }))
} finally {
  await browser.close()
  writeFileSync(join(ausgabe, 'montage-root-observer.json'), `${JSON.stringify({
    kind: 'ROOT_ADDED_PASSIVE_MONTAGE_OBSERVER',
    diagnostics: rootDiagnostik,
  }, null, 2)}\n`)
}
}

montage().catch((fehler: unknown) => {
  console.error(fehler)
  process.exitCode = 1
})
